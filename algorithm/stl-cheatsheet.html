二、顺序容器 ​

1. vector — 动态数组 ​

操作说明时间复杂度
vector<int> v(n)创建 n 个元素，默认值为 0O(n)
vector<int> v(n, val)创建 n 个元素，值为 valO(n)
v.push_back(x)尾部追加元素O(1)
v.pop_back()删除尾部元素O(1)
v[i]随机访问，不检查越界O(1)
v.at(i)随机访问，检查越界O(1)
v.front() / v.back()首 / 尾元素引用O(1)
v.size()元素个数O(1)
v.empty()是否为空O(1)
v.clear()清空O(n)
v.insert(it, x)在迭代器位置插入O(n)
v.erase(it)删除迭代器位置元素O(n)
v.resize(n)改变大小O(n)
v.reserve(n)预留容量O(n)
cpp
vector<int> v = {3, 1, 4, 1, 5};
sort(v.begin(), v.end());  // {1, 1, 3, 4, 5}
for (int x : v) cout << x << ' ';  // 范围遍历

2. string — 字符串 ​

操作说明时间复杂度
s.substr(pos, len)取子串O(len)
s.find(t, pos)查找子串 t，返回位置或 nposO(n·m)
s.compare(t)字典序比较O(n)
s += t追加字符串O(len)
s.push_back(c)追加单个字符O(1)
s.pop_back()删除末尾字符O(1)
to_string(x)数值 → 字符串O(log x)
stoi(s) / stoll(s)字符串 → int / long longO(len)
s.c_str()转 C 风格字符串O(1)
cpp
string s = "abc123";
int pos = s.find("123");       // 3
string num = s.substr(3, 3);   // "123"
int val = stoi(num);            // 123

3. deque — 双端队列 ​

操作说明复杂度
push_front(x) / push_back(x)首/尾插入O(1)
pop_front() / pop_back()首/尾删除O(1)
front() / back()首/尾元素O(1)
d[i]随机访问（比 vector 略慢）O(1)
cpp
deque<int> dq;
dq.push_back(2); dq.push_front(1);  // [1, 2]

4. list / forward_list — 双向/单向链表 ​

极少数情况用到（需要 O(1) 中间插入删除且无需随机访问时）。竞赛中 99% 用 vector 即可。

三、容器适配器 ​

5. stack — 栈（FILO） ​

cpp
stack<int> st;
st.push(x);       // 压栈 O(1)
st.pop();         // 弹栈 O(1)
st.top();         // 栈顶元素 O(1)
st.size();        // 大小 O(1)
st.empty();       // 是否空 O(1)

6. queue — 队列（FIFO） ​

cpp
queue<int> q;
q.push(x);        // 入队 O(1)
q.pop();          // 出队 O(1)
q.front();        // 队首 O(1)
q.back();         // 队尾 O(1)
q.size();         // 大小 O(1)

7. priority_queue — 优先队列（堆） ​

cpp
// 默认大顶堆
priority_queue<int> pq;
pq.push(x);       // 插入 O(log n)
pq.pop();         // 删除堆顶 O(log n)
pq.top();         // 堆顶（最大值）O(1)

// 小顶堆（两种写法）
priority_queue<int, vector<int>, greater<int>> min_pq;

// 自定义比较（大顶堆通常用 less，队尾传入的是"元素类型→容器→比较"）
struct Node { int v; };
struct cmp { bool operator()(Node& a, Node& b) { return a.v > b.v; } };
priority_queue<Node, vector<Node>, cmp> custom_pq;

四、关联容器 — 有序 ​

底层均为红黑树，操作 O(log n)。

8. set / multiset ​

操作说明
s.insert(x)插入（multiset 允许重复）
s.erase(x)删除值为 x 的所有元素（multiset 删全部）
s.erase(it)删除迭代器指向的单个元素
s.find(x)返回迭代器，找不到返回 s.end()
s.count(x)set：0 或 1；multiset：元素个数（O(log n + k)，慎用）
s.lower_bound(x)第一个 >= x 的迭代器
s.upper_bound(x)第一个 > x 的迭代器
s.begin() / s.rbegin()最小 / 最大元素
cpp
set<int> s = {3, 1, 4, 1};          // {1, 3, 4}，自动去重+排序
multiset<int> ms = {3, 1, 4, 1};    // {1, 1, 3, 4}

// multiset 删除一个元素（只删一个，不是全部）
auto it = ms.find(1);
if (it != ms.end()) ms.erase(it);    // 只删一个 1

9. map / multimap ​

操作说明
m[k]访问键 k，不存在则插入默认值
m.insert({k, v})插入键值对
m.erase(k)删除键
m.find(k)查找键
m.count(k)是否存在键（0 或 1）
m.lower_bound(k)第一个键 >= k 的迭代器
cpp
map<string, int> mp;
mp["abc"] = 1;                       // 插入或赋值
mp.insert({"def", 2});
if (mp.count("abc")) cout << mp["abc"];  // 安全查找

// 遍历
for (auto& [key, val] : mp)  // C++17 结构化绑定
    cout << key << ": " << val << '\n';

// 不存在键时不要用 []，用 find 避免意外插入
if (mp.find("xyz") != mp.end()) { /* 找到了 */ }

五、关联容器 — 无序（哈希） ​

底层为哈希表，平均 O(1)，最坏 O(n)。不支持 lower_bound / upper_bound，遍历顺序不确定。

cpp
unordered_set<int> us;
unordered_map<string, int> um;

us.insert(x);       // 平均 O(1)
um["key"] = val;    // 平均 O(1)

// 自定义哈希（以 pair<int,int> 为例）
struct pair_hash {
    size_t operator()(const pair<int,int>& p) const {
        return hash<long long>()(((long long)p.first << 32) | p.second);
    }
};
unordered_map<pair<int,int>, int, pair_hash> grid;

10. pair / tuple ​

11. bitset — 位集 ​

固定长度，位运算极快，常用于压位 DP / 优化布尔数组。

cpp
bitset<1000> b;           // 1000 位全 0
bitset<1000> b(42);       // 二进制 101010...
b.set(i);                 // 第 i 位置 1
b.reset(i);               // 第 i 位置 0
b.flip(i);                // 翻转第 i 位
b.test(i);                // 第 i 位是否为 1
b.count();                // 1 的个数 O(n/64)
b.any();                  // 是否有 1
b.none();                 // 是否全 0
b.all();                  // 是否全 1
b.to_ullong();            // 转 unsigned long long
b.to_string();            // 转 01 字符串

// 位运算
b1 & b2; b1 | b2; b1 ^ b2; ~b1;
b <<= k;                   // 左移 k 位

12. array — 定长数组（STL 封装） ​

cpp
array<int, 5> arr = {1, 2, 3, 4, 5};
arr.size();               // 5
arr.fill(0);              // 全部赋 0
sort(arr.begin(), arr.end());

七、<algorithm> 常用算法 ​

排序与排列 ​

cpp
// sort —— 默认升序，O(n log n)
sort(v.begin(), v.end());
sort(v.begin(), v.end(), greater<int>());   // 降序
sort(v.begin(), v.end(), [](int a, int b) { return a > b; });  // lambda 自定义

// stable_sort —— 稳定排序，相等元素保持原顺序
stable_sort(v.begin(), v.end());

// partial_sort —— 部分排序，v.begin()+k 之前是最小的 k 个
partial_sort(v.begin(), v.begin() + k, v.end());

// nth_element —— 第 k 小元素归位，前后不保证有序，O(n)
nth_element(v.begin(), v.begin() + k, v.end());
// 第 k 小的元素在 v[k]，左边都 ≤ v[k]，右边都 ≥ v[k]

// next_permutation —— 下一个字典序排列，返回 false 表示已是最大排列
do {
    // 处理当前排列
} while (next_permutation(v.begin(), v.end()));
// 注意：需要先 sort 才能遍历全部排列

// prev_permutation —— 上一个字典序排列
prev_permutation(v.begin(), v.end());

// is_sorted —— 是否已排序
is_sorted(v.begin(), v.end());

// reverse —— 翻转
reverse(v.begin(), v.end());

// rotate —— 左旋，middle 成为新首元素
rotate(v.begin(), v.begin() + k, v.end());
// [0,k) 被移到末尾，[k,end) 移到开头

查找 ​

cpp
// find —— 线性查找，返回迭代器或末尾
auto it = find(v.begin(), v.end(), x);

// find_if —— 按条件查找
auto it = find_if(v.begin(), v.end(), [](int x) { return x > 10; });

// count —— 计数
int cnt = count(v.begin(), v.end(), x);

// count_if —— 按条件计数
int cnt = count_if(v.begin(), v.end(), [](int x) { return x % 2 == 0; });

// binary_search —— 二分查找是否存在，O(log n)
bool exists = binary_search(v.begin(), v.end(), x);   // 要求已排序

// lower_bound —— 第一个 ≥ x 的位置，O(log n)
auto it = lower_bound(v.begin(), v.end(), x);
int idx = it - v.begin();          // 下标
// 如果全小于 x，返回 v.end()

// upper_bound —— 第一个 > x 的位置，O(log n)
auto it = upper_bound(v.begin(), v.end(), x);

// equal_range —— 返回 [lower_bound, upper_bound) 的 pair
auto [l, r] = equal_range(v.begin(), v.end(), x);
int count = r - l;                  // 区间内元素个数

// max_element / min_element —— 最大 / 最小元素的迭代器
auto it = max_element(v.begin(), v.end());
int mx = *max_element(v.begin(), v.end());

// minmax_element —— 同时返回最小和最大，只需一次遍历
auto [mn, mx] = minmax_element(v.begin(), v.end());

修改序列 ​

cpp
// fill —— 全部赋值
fill(v.begin(), v.end(), 0);

// iota —— 递增填充
iota(v.begin(), v.end(), 0);       // {0, 1, 2, ..., n-1}

// copy —— 复制
copy(src.begin(), src.end(), back_inserter(dst));      // 尾部追加
copy(src.begin(), src.end(), dst.begin());              // 覆盖（dst 需够大）

// unique —— 将连续重复元素移到末尾，返回新"末尾"迭代器
// 通常搭配 erase 做真正去重（需先排序）
sort(v.begin(), v.end());
v.erase(unique(v.begin(), v.end()), v.end());

// 离散化模板
sort(v.begin(), v.end());
v.erase(unique(v.begin(), v.end()), v.end());
// 用 lower_bound 查询离散化后的值
int discrete_val = lower_bound(v.begin(), v.end(), x) - v.begin();

// remove —— 将等于 x 的元素移到末尾
auto new_end = remove(v.begin(), v.end(), x);
v.erase(new_end, v.end());          // 真正删除

// remove_if —— 按条件移除
v.erase(remove_if(v.begin(), v.end(), [](int x) { return x < 0; }), v.end());

// replace —— 替换
replace(v.begin(), v.end(), old_val, new_val);

// transform —— 对每个元素应用函数，结果写入目标
transform(v.begin(), v.end(), res.begin(), [](int x) { return x * 2; });

// swap —— 交换两个变量
swap(a, b);

// iter_swap —— 交换迭代器指向的元素
iter_swap(v.begin(), v.begin() + 1);

数值操作（<numeric>） ​

cpp
// accumulate —— 累加，第三个参数为初始值
int sum = accumulate(v.begin(), v.end(), 0);
// 注意：初始值 0 是 int，如果求 long long 要写 0LL

// 累乘
long long prod = accumulate(v.begin(), v.end(), 1LL, multiplies<long long>());

// partial_sum —— 前缀和，结果写入目标
vector<int> pref(n);
partial_sum(v.begin(), v.end(), pref.begin());

// adjacent_difference —— 差分（相邻差）
vector<int> diff(n);
adjacent_difference(v.begin(), v.end(), diff.begin());

// gcd / lcm（C++17）
gcd(a, b);              // 最大公约数
lcm(a, b);              // 最小公倍数
// 注意：C++14 用 __gcd(a, b)（GCC 扩展）

// inner_product —— 内积
int dot = inner_product(v1.begin(), v1.end(), v2.begin(), 0);

集合操作（要求已排序） ​

cpp
// 结果用 back_inserter 追加到目标容器
vector<int> res;

// set_union —— 并集
set_union(a.begin(), a.end(), b.begin(), b.end(), back_inserter(res));

// set_intersection —— 交集
set_intersection(a.begin(), a.end(), b.begin(), b.end(), back_inserter(res));

// set_difference —— 差集（a \ b，在 a 但不在 b）
set_difference(a.begin(), a.end(), b.begin(), b.end(), back_inserter(res));

// set_symmetric_difference —— 对称差（在 a 或 b，但不同时在）
set_symmetric_difference(a.begin(), a.end(), b.begin(), b.end(), back_inserter(res));

// includes —— a 是否包含 b 的所有元素
bool ok = includes(a.begin(), a.end(), b.begin(), b.end());

// merge —— 合并两个有序序列
merge(a.begin(), a.end(), b.begin(), b.end(), back_inserter(res));

// inplace_merge —— 原地合并（v 两端各自有序）
inplace_merge(v.begin(), v.begin() + mid, v.end());

堆操作 ​

cpp
// 直接在 vector 上操作堆，不用 priority_queue
vector<int> v;

make_heap(v.begin(), v.end());            // 建堆 O(n)
push_heap(v.begin(), v.end());            // v.push_back 后调用，维护堆
pop_heap(v.begin(), v.end());             // 堆顶移到末尾，v.pop_back 前调用
sort_heap(v.begin(), v.end());            // 堆排序 O(n log n)

// 小顶堆
make_heap(v.begin(), v.end(), greater<int>());

八、迭代器辅助 ​

cpp
// advance —— 迭代器前进 n 步
advance(it, n);

// distance —— 两个迭代器间的距离
int d = distance(v.begin(), it);

// next / prev（C++11）
auto it2 = next(it, 3);    // it 前进 3 步后的迭代器
auto it3 = prev(it, 2);    // it 后退 2 步

// 四种反向迭代器
v.rbegin();   v.rend();     // 反向遍历
v.cbegin();   v.cend();     // const 正向
v.crbegin();  v.crend();    // const 反向

// 插入迭代器
back_inserter(v);            // push_back
front_inserter(dq);          // push_front（仅 deque/list）
inserter(s, s.begin());      // insert（set/map 等）
